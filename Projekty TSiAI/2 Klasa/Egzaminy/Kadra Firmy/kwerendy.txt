FROM pracownicy p
GROUP BY p.pensja

SELECT p.imie, p.nazwisko, s.nazwa
FROM pracownicy p
JOIN stanowiska s
ON p.stanowiska_id = s.id
WHERE p.staz > 10;

FROM pracownicy p
WHERE p.staz BETWEEN 10 AND 20;
