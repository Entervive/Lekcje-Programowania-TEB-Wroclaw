FROM zdjecia z

2.
SELECT z.plik, z.tytul, z.polubienia, a.imie, a.nazwisko
FROM zdjecia z
JOIN autorzy a
ON z.autorzy_id = a.id
ORDER BY a.nazwisko ASC;

3.
SELECT a.imie, COUNT(z.autorzy_id)
FROM autorzy a
JOIN zdjecia z
ON a.id = z.autorzy_id
GROUP BY a.imie;
